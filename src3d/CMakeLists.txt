function(rs_options target)
  target_compile_options(${target} PRIVATE /W4 /EHsc /fp:precise /utf-8)
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endfunction()

set(RS3D_HASH_INPUT "")
foreach(path IN LISTS RS3D_SOURCES)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(SHA256 "${path}" digest)
  file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}" "${path}")
  string(APPEND RS3D_HASH_INPUT "${relative}:${digest}\n")
endforeach()
string(SHA256 RS3D_SOURCE_HASH "${RS3D_HASH_INPUT}")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE RS3D_SOURCE_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE)
